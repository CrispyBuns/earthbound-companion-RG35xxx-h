cmake_minimum_required(VERSION 3.16)
project(CompanionAchievements VERSION 0.1.0 LANGUAGES C)
add_library(companion_achievements STATIC src/companion_achievements.c)
target_include_directories(companion_achievements PUBLIC include)
target_compile_features(companion_achievements PUBLIC c_std_99)
include(CTest)
if(BUILD_TESTING)
  add_executable(core_test tests/core_test.c)
  target_link_libraries(core_test PRIVATE companion_achievements)
  add_test(NAME core COMMAND core_test)
endif()
